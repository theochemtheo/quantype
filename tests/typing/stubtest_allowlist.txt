# Python 3.12 generic runtime metadata is not represented by mypy's PEP 695 stubs.
# PEP 695 cannot declare the defaulted unit-system parameter before Python 3.13,
# so the stubs declare typing_extensions TypeVars that the runtime has no use for.
^quantype\.(core|_generated)\.(K|V|S)$
^quantype\._generated\.[A-Za-z]+\.__type_params__$
^quantype\.core\.Quantity\.__type_params__$
^quantype\.core\.Constant\.__type_params__$
^quantype\._internal\._unit\.Unit\.__type_params__$
^quantype\._internal\._kind_types\.(Div|Mul|Pow)\.__type_params__$

# A type alias has no runtime tuple base class (stubtest treats it as a class).
^quantype\._internal\._registry\.Dimensions$
^quantype\._internal\._semantics\.Semantic$
^quantype\._internal\._codata\.Edition$
^quantype\._internal\._registry\.Relations$

# functools.cache exposes a variadic checker signature but preserves runtime __wrapped__.
^quantype\._internal\._registry\._unit_specs$

# Standard static generics are retained; the runtime alias adds conversion on call.
^quantype\.core\.Quantity\.__class_getitem__$

# NumPy coercion/dispatch methods intentionally reject implicit unit loss.
^quantype\.core\.Quantity\.(__array__|__array_function__|__array_priority__|__array_ufunc__)$

# Unit calls/operators have kind-specific overloads on the generated unit types.
# NumPy dispatch and array priority are runtime-only implementation details.
^quantype\._internal\._unit\.Unit\.(__array_priority__|__array_ufunc__|__call__|__mul__|__rmul__)$

# The unit catalogue's stub variables use kind-specific static subclasses of
# Unit so unit calls return named quantities; runtime instances are plain Unit.
^quantype\.units\.(?!_)(?!(exp|sin|sqrt)$)[A-Za-z0-9_]+$

# UnitSystem subclasses must pass name=, which stubtest's probe subclass omits.
^quantype\._internal\._systems\.(UnitSystem|Atomistic|Metal|Real|SI|CGS|Atomic)$
# The static signature requires name=; the runtime default gives a clearer error.
^quantype\._internal\._systems\.UnitSystem\.__init_subclass__$
# A descriptor: each concrete system computes its own units on access.
^quantype\._internal\._systems\.UnitSystem\.units$
